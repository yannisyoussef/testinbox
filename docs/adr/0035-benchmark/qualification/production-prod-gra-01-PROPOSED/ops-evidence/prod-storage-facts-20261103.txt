# /etc/systemd/system/testinbox-minio-fs-check.service
# /etc/systemd/system/testinbox-minio-fs-check.service (TI-008 D3) — re-verify the dedicated object filesystem:
# image allocation (FIEMAP, no holes), exact loop/inode identity, UUID, discard_max_bytes=0, rw,relatime, geometry.
# Writes /run/infinity/testinbox-minio-fs/status (read by the deploy preflight gate) and the node-exporter textfile
# testinbox-minio-fs.prom. Read-only towards the filesystem: it never attaches, mounts or repairs anything.
[Unit]
Description=TestInbox dedicated MinIO filesystem check
After=testinbox-minio-fs.service
ConditionPathExists=/etc/infinity/testinbox-minio-fs.env

[Service]
Type=oneshot
ExecStart=/usr/local/sbin/testinbox-minio-fs check
TimeoutStartSec=120

# /etc/systemd/system/testinbox-minio-fs-check.timer
# /etc/systemd/system/testinbox-minio-fs-check.timer (TI-008 D3) — every 5 minutes, UTC (house rule for sub-daily
# timers). The deploy gate needs a PASS no older than 15 minutes.
[Unit]
Description=TestInbox dedicated MinIO filesystem check (every 5 minutes)

[Timer]
OnCalendar=*-*-* *:00/5:00 UTC
OnBootSec=1min
AccuracySec=10s

[Install]
WantedBy=timers.target

# /etc/systemd/system/testinbox-minio-fs.service
# /etc/systemd/system/testinbox-minio-fs.service (TI-008 D3) — attach and mount TestInbox's dedicated object filesystem.
# A supported, ordered oneshot (not a .mount unit: a mount unit has no ExecStartPost=, so a discard guard there would
# never run). ExecStart verifies the recorded image (ownership, full allocation, UUID), attaches THAT inode to a loop
# device, sets and proves discard_max_bytes=0, mounts rw,relatime and verifies the live mount; any failure leaves the
# filesystem unmounted, so TestInbox's MinIO bind (create_host_path: false) refuses to start.
#
# Ordering only: Before=docker.service lets the restart policy find the mount at boot. Nothing here, and no drop-in,
# makes Docker REQUIRE this unit (no Requires=/BindsTo=/RequiresMountsFor= on docker.service): if this unit fails,
# Docker and every other stack start normally and only TestInbox's MinIO stays down (D3 review D2).
[Unit]
Description=TestInbox dedicated MinIO filesystem (loop attach, discard guard, verified mount)
After=local-fs.target
Before=docker.service
ConditionPathExists=/etc/infinity/testinbox-minio-fs.env

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/usr/local/sbin/testinbox-minio-fs attach
ExecStop=/usr/local/sbin/testinbox-minio-fs detach
TimeoutStartSec=120
TimeoutStopSec=120

[Install]
WantedBy=multi-user.target
